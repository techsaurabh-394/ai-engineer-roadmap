# Put this in the .htaccess of the web root (mernstackdev.com), ABOVE "# BEGIN WordPress".
<IfModule mod_rewrite.c>
RewriteEngine On

# 1. Old location -> new location, permanent, path kept (/ai-roadmap/learn/x/ -> /ai-engineer-roadmap/learn/x/)
RewriteRule ^ai-roadmap/?(.*)$ /ai-engineer-roadmap/$1 [R=301,L]

# 2. Serve the roadmap's real files and folders directly, so WordPress never rewrites them
RewriteCond %{REQUEST_URI} ^/ai-engineer-roadmap/
RewriteCond %{REQUEST_FILENAME} -f [OR]
RewriteCond %{REQUEST_FILENAME} -d
RewriteRule ^ - [L]
</IfModule>

# Static files: long cache for the versioned assets, short for HTML
<IfModule mod_headers.c>
<FilesMatch "\.(css|js|svg|png)$">
Header set Cache-Control "public, max-age=31536000, immutable"
</FilesMatch>
<FilesMatch "\.(html|xml|txt)$">
Header set Cache-Control "public, max-age=3600"
</FilesMatch>
</IfModule>
